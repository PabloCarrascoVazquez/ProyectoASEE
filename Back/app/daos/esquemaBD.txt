
CREATE TABLE IF NOT EXISTS events (
    id VARCHAR(36) PRIMARY KEY,
    name VARCHAR(150) NOT NULL,
    date DATETIME NOT NULL,
    image VARCHAR(255) NULL,
    description TEXT NOT NULL,
    services JSON NULL,       
    gallery JSON NULL,        
    mapImage VARCHAR(255) NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Tabla Artists
CREATE TABLE IF NOT EXISTS artists (
    id VARCHAR(36) PRIMARY KEY,
    name VARCHAR(150) NOT NULL,
    genre VARCHAR(100) NOT NULL,
    image VARCHAR(255) NULL,
    spotifyUrl VARCHAR(255) NULL
) ENGINE=InnoDB;

-- Tabla Merchandise (Catálogo global)
CREATE TABLE IF NOT EXISTS merchandise_items (
    id VARCHAR(36) PRIMARY KEY,
    name VARCHAR(150) NOT NULL,
    price DECIMAL(10,2) NOT NULL, 
    image VARCHAR(255) NULL,
    category VARCHAR(100) NOT NULL
) ENGINE=InnoDB;

-- ==========================================
-- 2. TABLAS DEPENDIENTES (Con Claves Foráneas)
-- ==========================================

-- Tabla TicketPrices (Tipos de entrada disponibles para un evento)
CREATE TABLE IF NOT EXISTS ticket_prices (
    id VARCHAR(36) PRIMARY KEY,
    event_id VARCHAR(36) NOT NULL,
    type VARCHAR(50) NOT NULL,                   -- Ej: 'VIP', 'General', 'Early Bird'
    price DECIMAL(10,2) NOT NULL,
    description TEXT NULL,
    features JSON NULL,
    CONSTRAINT fk_ticket_event 
        FOREIGN KEY (event_id) REFERENCES events(id) 
        ON DELETE CASCADE
) ENGINE=InnoDB;

-- Tabla VendorSpaces 
CREATE TABLE IF NOT EXISTS vendor_spaces (
    id VARCHAR(36) PRIMARY KEY,
    event_id VARCHAR(36) NOT NULL,
    name VARCHAR(150) NOT NULL,
    type VARCHAR(100) NOT NULL,
    size VARCHAR(50) NOT NULL,
    location VARCHAR(150) NOT NULL,
    price DECIMAL(10,2) NOT NULL,
    services JSON NULL, 
    image VARCHAR(255) NULL,
    dimensions VARCHAR(100) NOT NULL,
    available BOOLEAN DEFAULT TRUE,
    CONSTRAINT fk_vendor_event 
        FOREIGN KEY (event_id) REFERENCES events(id) 
        ON DELETE CASCADE
) ENGINE=InnoDB;

-- ==========================================
-- 3. TABLAS TRANSACCIONALES Y DE RELACIÓN
-- ==========================================

-- NUEVO: Tabla UserTickets (Entradas compradas por el usuario)
-- Representa una instancia física/digital de una entrada que posee un usuario
CREATE TABLE IF NOT EXISTS user_tickets (
    id VARCHAR(36) PRIMARY KEY,                  -- ID único de esta entrada (sirve para generar el QR)
    user_id VARCHAR(36) NOT NULL,                -- Quién la compró
    ticket_price_id VARCHAR(36) NOT NULL,        -- Qué tipo de entrada es
    purchase_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    status VARCHAR(50) DEFAULT 'ACTIVE',         -- Estados lógicos: 'ACTIVE', 'USED', 'REFUNDED'
    CONSTRAINT fk_ut_user 
        FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    CONSTRAINT fk_ut_ticket_price 
        FOREIGN KEY (ticket_price_id) REFERENCES ticket_prices(id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- Relación Muchos-a-Muchos entre Events y Artists
CREATE TABLE IF NOT EXISTS event_artists (
    event_id VARCHAR(36) NOT NULL,
    artist_id VARCHAR(36) NOT NULL,
    PRIMARY KEY (event_id, artist_id), 
    CONSTRAINT fk_ea_event 
        FOREIGN KEY (event_id) REFERENCES events(id) ON DELETE CASCADE,
    CONSTRAINT fk_ea_artist 
        FOREIGN KEY (artist_id) REFERENCES artists(id) ON DELETE CASCADE
) ENGINE=InnoDB;
