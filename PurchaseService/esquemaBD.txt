-- Base de datos dedicada para el microservicio: purchases_db
CREATE DATABASE IF NOT EXISTS purchases_db;
USE purchases_db;

-- ==========================================
-- 1. TABLA PRINCIPAL: TRANSACCIONES / COMPRAS
-- ==========================================
CREATE TABLE IF NOT EXISTS transactions (
    id VARCHAR(36) PRIMARY KEY,                         -- UUID de la transacción (id_transaccion)
    user_id VARCHAR(36) NOT NULL,                       -- ID del usuario comprador
    type ENUM('ticket', 'merchandise') NOT NULL,        -- Compra de entrada o merchandising
    total_amount DECIMAL(10, 2) NOT NULL,               -- Importe total a cobrar
    currency VARCHAR(3) NOT NULL DEFAULT 'EUR',
    
    -- Control del ciclo de vida transaccional
    status ENUM('PENDING', 'PAID', 'EXPIRED', 'FAILED', 'CANCELLED') NOT NULL DEFAULT 'PENDING',
    reservation_expires_at DATETIME NOT NULL,           -- Límite del bloqueo temporal (ej. 10 minutos)
    
    -- Datos de la pasarela de pagos
    payment_method ENUM('tarjeta', 'paypal', 'bizum') NULL,
    gateway_token VARCHAR(255) NULL,                    -- Token de Stripe/Redsys
    gateway_transaction_id VARCHAR(255) NULL,           -- ID de confirmación bancaria
    receipt_url VARCHAR(500) NULL,                      -- Comprobante emitido por la pasarela
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    INDEX idx_user_id (user_id),
    INDEX idx_status_expires (status, reservation_expires_at)
) ENGINE=InnoDB;

-- ==========================================
-- 2. TABLA DETALLE: LÍNEAS DEL CARRITO / RESERVAS
-- ==========================================
CREATE TABLE IF NOT EXISTS transaction_items (
    id VARCHAR(36) PRIMARY KEY,
    transaction_id VARCHAR(36) NOT NULL,
    item_type ENUM('ticket', 'merchandise') NOT NULL,
    item_id VARCHAR(36) NOT NULL,                       -- id_festival o id_producto
    ticket_type ENUM('day', 'weekend', 'vip') NULL,     -- Aplica solo a entradas
    quantity INT NOT NULL DEFAULT 1,
    unit_price DECIMAL(10, 2) NOT NULL,
    subtotal DECIMAL(10, 2) NOT NULL,
    
    CONSTRAINT fk_items_transaction
        FOREIGN KEY (transaction_id) 
        REFERENCES transactions(id) 
        ON DELETE CASCADE
) ENGINE=InnoDB;

-- ==========================================
-- 3. TABLA DE EVENTOS PUBLICADOS (PATRÓN TRANSACTIONAL OUTBOX)
-- ==========================================
-- Guarda los eventos pendientes de emitir a Kafka/RabbitMQ tras confirmar el pago
CREATE TABLE IF NOT EXISTS outbox_events (
    id VARCHAR(36) PRIMARY KEY,
    aggregate_type VARCHAR(50) NOT NULL DEFAULT 'TRANSACTION',
    aggregate_id VARCHAR(36) NOT NULL,                  -- transaction_id
    event_type VARCHAR(50) NOT NULL,                    -- 'ORDER_PAID', 'RESERVATION_EXPIRED'
    payload JSON NOT NULL,                              -- Datos que consumen Pedidos e Inventario
    published BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    INDEX idx_unpublished (published, created_at)
) ENGINE=InnoDB;
