-- =============================================================================

-- ESQUEMA DE BASE DE DATOS: SERVICIO DE INVENTARIOS Y ENVÍOS

-- Base de datos independiente (Patrón Database-per-Service)

-- Sin claves foráneas (FK) directas a tablas de otros microservicios;

-- las referencias externas (id_producto, id_pedido, id_usuario) se validan vía HTTP.

-- =============================================================================



CREATE DATABASE IF NOT EXISTS db_inventario_envios;

USE db_inventario_envios;



-- -----------------------------------------------------------------------------

-- 1. Tabla: inventario_merchandising

-- Almacena el stock físico de los productos de merchandising en almacén.

-- -----------------------------------------------------------------------------

CREATE TABLE inventario_merchandising (

    id_producto BIGINT PRIMARY KEY,              -- ID externo proveniente del servicio de Catálogo/Productos

    nombre_producto VARCHAR(150) NOT NULL,       -- Nombre cacheado/descriptivo del ítem

    stock_disponible INT NOT NULL DEFAULT 0,     -- Cantidad física en almacén (No negativa)

    ubicacion_almacen VARCHAR(100),              -- Ej: "Pasillo A - Estante 3"

    fecha_actualizacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT chk_stock_positivo CHECK (stock_disponible >= 0)

);



-- -----------------------------------------------------------------------------

-- 2. Tabla: envios

-- Gestiona las órdenes logísticas y el tracking de paquetes.

-- -----------------------------------------------------------------------------

CREATE TABLE envios (

    id_envio BIGINT AUTO_INCREMENT PRIMARY KEY,

    id_pedido BIGINT NOT NULL,                   -- ID externo del pedido (Servicio de Ventas/Pedidos)

    id_usuario BIGINT NOT NULL,                  -- ID externo del cliente (UserService)

    direccion_destino VARCHAR(255) NOT NULL,

    estado_logistico ENUM('PENDIENTE', 'EN_PREPARACION', 'EN_TRANSITO', 'ENTREGADO', 'INCIDENCIA') NOT NULL DEFAULT 'PENDIENTE',

    codigo_tracking VARCHAR(100) UNIQUE,

    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    fecha_actualizacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP

);



-- Índices para optimizar las consultas de los endpoints

CREATE INDEX idx_envios_estado ON envios(estado_logistico);

CREATE INDEX idx_envios_pedido ON envios(id_pedido);
