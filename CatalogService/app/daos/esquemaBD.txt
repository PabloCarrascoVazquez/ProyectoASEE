-- =============================================================================
-- ESQUEMA DE BASE DE DATOS: SERVICIO DE CATÁLOGO (CatalogService)
-- Base de datos independiente (Patrón Database-per-Service)
-- Soporta SQLite y MySQL
-- =============================================================================

-- MySQL:
-- CREATE DATABASE IF NOT EXISTS catalogo_db;
-- USE catalogo_db;

-- -----------------------------------------------------------------------------
-- 1. Tabla: Festivales
-- Agrupa festivales musicales, fechas, ubicaciones y escaleta de artistas
-- -----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS Festivales (
    id_festival VARCHAR(50) PRIMARY KEY,
    nombre VARCHAR(150) NOT NULL,
    fecha_inicio DATE NOT NULL,
    fecha_fin DATE NOT NULL,
    ubicacion VARCHAR(255) NOT NULL,
    artistas_destacados TEXT NULL -- Formato JSON, ej: '["Daft Punk", "Carl Cox"]'
);

-- -----------------------------------------------------------------------------
-- 2. Tabla: Espacios
-- Parcelas y puestos disponibles dentro del recinto del festival
-- -----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS Espacios (
    id_espacio VARCHAR(36) PRIMARY KEY,
    id_festival VARCHAR(50) NOT NULL,
    tipo VARCHAR(50) NOT NULL, -- foodtruck, mercadillo, vip_zone
    metros_cuadrados INT NOT NULL,
    precio_alquiler DECIMAL(10, 2) NOT NULL,
    estado VARCHAR(20) NOT NULL DEFAULT 'disponible', -- disponible, reservado, alquilado
    FOREIGN KEY (id_festival) REFERENCES Festivales(id_festival)
);

-- -----------------------------------------------------------------------------
-- 3. Tabla: Alquileres
-- Contratos y solicitudes de alquiler de parcelas por parte de proveedores
-- -----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS Alquileres (
    id_contrato VARCHAR(36) PRIMARY KEY,
    id_espacio VARCHAR(36) NOT NULL,
    nombre_negocio VARCHAR(100) NOT NULL,
    tipo_negocio VARCHAR(100) NOT NULL,
    necesita_conexion_electrica BOOLEAN NOT NULL DEFAULT 0,
    estado VARCHAR(50) NOT NULL DEFAULT 'Pendiente de firma',
    fecha_solicitud DATETIME NOT NULL,
    FOREIGN KEY (id_espacio) REFERENCES Espacios(id_espacio)
);

-- -----------------------------------------------------------------------------
-- 4. Tabla: Tipos_Entrada
-- Tipos de abonos y entradas disponibles por festival con tarifas y cupo
-- -----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS Tipos_Entrada (
    id_festival VARCHAR(50) NOT NULL,
    tipo VARCHAR(20) NOT NULL, -- day, weekend, vip
    precio DECIMAL(10, 2) NOT NULL,
    disponibles INT NOT NULL DEFAULT 0,
    PRIMARY KEY (id_festival, tipo),
    FOREIGN KEY (id_festival) REFERENCES Festivales(id_festival)
);

-- -----------------------------------------------------------------------------
-- 5. Tabla: Artistas_Favoritos
-- Artistas guardados como favoritos por cada usuario en un festival concreto
-- -----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS Artistas_Favoritos (
    user_id VARCHAR(36) NOT NULL,
    id_festival VARCHAR(50) NOT NULL,
    nombre_artista VARCHAR(100) NOT NULL,
    PRIMARY KEY (user_id, id_festival, nombre_artista),
    FOREIGN KEY (id_festival) REFERENCES Festivales(id_festival)
);

-- -----------------------------------------------------------------------------
-- 6. Tabla: Artistas
-- Catálogo general de artistas con géneros y niveles de popularidad
-- -----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS Artistas (
    id INTEGER PRIMARY KEY AUTOINCREMENT, -- en MySQL: INT AUTO_INCREMENT PRIMARY KEY
    nombre VARCHAR(100) NOT NULL,
    genero VARCHAR(50),
    popularidad INT DEFAULT 0
);
