-- =====================================================================
-- Esquema de Base de Datos - OrderService (Wallet de Entradas y Acceso)
-- Compatible con SQLite y PostgreSQL
-- =====================================================================

-- Tabla de Entradas Digitales (Wallet)
CREATE TABLE IF NOT EXISTS entradas (
    id_entrada VARCHAR(36) PRIMARY KEY,
    id_festival VARCHAR(100) NOT NULL,
    user_id VARCHAR(36) NOT NULL,
    tipo VARCHAR(20) NOT NULL,          -- 'General', 'VIP', 'Abono'
    fecha_compra TIMESTAMP NOT NULL,
    estado VARCHAR(20) NOT NULL DEFAULT 'Valida', -- 'Valida', 'Usada', 'Cancelada'
    transaccion_id VARCHAR(100) NULL,   -- ID de transacción de PurchaseService (idempotencia)
    fecha_uso TIMESTAMP NULL            -- Fecha y hora del escaneo en puerta
);

CREATE INDEX IF NOT EXISTS idx_entradas_user_id ON entradas(user_id);
CREATE INDEX IF NOT EXISTS idx_entradas_transaccion_id ON entradas(transaccion_id);

-- Tabla de Registro de Cancelaciones / Devoluciones
CREATE TABLE IF NOT EXISTS cancelaciones (
    id_cancelacion VARCHAR(36) PRIMARY KEY,
    id_entrada VARCHAR(36) NOT NULL,
    user_id VARCHAR(36) NOT NULL,
    estado_devolucion VARCHAR(50) NOT NULL DEFAULT 'En proceso', -- 'En proceso', 'Completada', 'Rechazada'
    fecha_solicitud TIMESTAMP NOT NULL,
    motivo VARCHAR(255) NULL,
    FOREIGN KEY (id_entrada) REFERENCES entradas(id_entrada)
);

CREATE INDEX IF NOT EXISTS idx_cancelaciones_entrada ON cancelaciones(id_entrada);
CREATE INDEX IF NOT EXISTS idx_cancelaciones_user ON cancelaciones(user_id);
